{{- /* The house ampersand, for content that is ALREADY HTML (rendered .Content). Same two patterns as rich.html, which starts from markdown instead: freestanding "&" (and "&c."), and letter-tight ampersands between capitals (Texas A&M, ASIS&T). */ -}} {{- $html := printf "%s" . -}} {{- $html = replaceRE `(^|[\s(>])&(\s|c\.|$)` `$1&$2` $html -}} {{- $html = replaceRE `([A-Z])&([A-Z])` `$1&$2` $html -}} {{- $html | safeHTML -}}
